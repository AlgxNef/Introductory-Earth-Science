import { 
  saveSvg,
  saveSvgAsPng,
  svgAsPngUri,
} from 'save-svg-as-png';

const siteName = "理系地学入門 - SCIENCE to IMG";
const siteUrl = "https://i-earth.pages.dev";
	
import katexCss from 'katex/dist/katex.min.css';

  const newCm10FontFamily = 'newcm10_36c61f'; 
  const newCmMathFontFamily = 'newcmmath_3be6b6';
	
// save-svg-as-pngのオプションを設定
const getOptions = (svgElement: SVGSVGElement, excludedElements: Element[], format?: 'svg' | 'png' | 'jpeg' | 'webp') => {
  const viewBox = svgElement.viewBox.baseVal;
  const width = viewBox.width;
  const height = viewBox.height;
  // SVG内のKaTeXフォントをインライン化するための設定
  const katexFontRegex = /@font-face\s*{[^}]*font-family:\s*KaTeX_Main[^}]*}/g;
  
  const options: any = {
    exclude: excludedElements,
    width: width,
    height: height,
    left: viewBox.x,
    top: viewBox.y,
    modifyStyle: (styleText: string) => {
      const comment = `/* Generated by ${siteName} (${siteUrl}) */`;
      // ライブラリが自動で集めたスタイル(styleText)は無視し、
      // KaTeXのCSSとクレジットコメントだけを返す
      return `${comment}\n${katexCss}`;
    },
  };	
  return options;
};

// --- SVGエクスポート関数 ---
export const exportAsSVG = (svgElement: SVGSVGElement, filename: string, excludedElements: Element[]) => {
  const options = getOptions(svgElement, excludedElements, 'svg'); 
  const comment = document.createComment(
    `\n  Generated by ${siteName} (${siteUrl})\n  Date: ${new Date().toISOString()}\n`
  );
  svgElement.insertBefore(comment, svgElement.firstChild);
  saveSvg(svgElement, `${filename}.svg`, options);
	svgElement.removeChild(comment);
};

// --- ラスタ画像エクスポート関数 ---
export const exportAsRaster = (
  svgElement: SVGSVGElement, 
  filename: string, 
  format: 'png' | 'jpeg' | 'webp',
  excludedElements: Element[]
) => {
  const options = getOptions(svgElement, excludedElements, format);
  
	if (format === 'png') {
    saveSvgAsPng(svgElement, `${filename}.png`, options);
    return;
  }
	
  svgAsPngUri(svgElement, options).then((uri: string) => {
    const img = new Image();
    img.onload = () => {
      const canvas = document.createElement('canvas');
      canvas.width = img.width;
      canvas.height = img.height;
      const ctx = canvas.getContext('2d');
      if (!ctx) return;
      if (format === 'jpeg') {
        ctx.fillStyle = 'white';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
      }

      ctx.drawImage(img, 0, 0);
      canvas.toBlob(
        (blob) => {
          if (blob) {
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `${filename}.${format}`;
            link.click();
            URL.revokeObjectURL(link.href);
          }
        },
        `image/${format}`
      );
    };
    img.src = uri;
  });
};
